import React from "react";
import { Icon } from "../core/Icon.jsx";

export function SearchField({ value = "", placeholder = "Para onde?", focused = false, onChange, onClear, onBack, style, ...rest }) {
  return (
    <div style={{ display: "flex", alignItems: "center", gap: "8px", ...style }} {...rest}>
      {onBack && (
        <button type="button" aria-label="Voltar" onClick={onBack} style={{ background: "none", border: "none", color: "var(--text-primary)", cursor: "pointer", padding: "8px", display: "flex" }}>
          <Icon name="arrow-left" size={24} />
        </button>
      )}
      <div
        style={{
          flex: 1, display: "flex", alignItems: "center", gap: "10px", height: "52px", padding: "0 14px",
          background: "var(--surface-card)", borderRadius: "var(--radius-md)",
          border: focused ? "var(--border-width-selected) solid var(--border-selected)" : "var(--border-width) solid var(--border-hairline)",
          transition: "border-color var(--dur-base) var(--ease-standard)",
        }}
      >
        <span style={{ width: 10, height: 10, borderRadius: "var(--radius-pill)", background: "var(--accent-primary)", flex: "0 0 auto" }} />
        <input
          value={value}
          placeholder={placeholder}
          onChange={onChange}
          style={{
            flex: 1, minWidth: 0, background: "transparent", border: "none", outline: "none",
            color: "var(--text-primary)", fontFamily: "var(--font-core)", fontSize: "17px", fontWeight: "var(--fw-bold)",
          }}
        />
        {value && onClear && (
          <button type="button" aria-label="Limpar" onClick={onClear} style={{ background: "none", border: "none", color: "var(--text-tertiary)", cursor: "pointer", display: "flex", padding: 0 }}>
            <Icon name="circle-x" size={22} />
          </button>
        )}
      </div>
    </div>
  );
}
