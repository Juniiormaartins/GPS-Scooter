import React from "react";
import { Icon } from "../core/Icon.jsx";

const TILE_TONES = {
  neutral: { background: "var(--surface-tile)", color: "var(--text-secondary)" },
  accent: { background: "var(--blue-a16)", color: "var(--accent-primary)" },
  warn: { background: "var(--surface-tile)", color: "var(--accent-warn)" },
  go: { background: "var(--surface-tile)", color: "var(--accent-go)" },
};

/* The universal list row: icon tile, title, subtitle, and a trailing value or chevron.
   Covers search results, saved places and ride history in the source screens. */
export function ListRow({ icon, iconShape = "square", tone = "neutral", title, subtitle, trailing, chevron = false, divider = false, onClick, style, ...rest }) {
  return (
    <div
      onClick={onClick}
      style={{
        display: "flex", alignItems: "center", gap: "16px", minHeight: "var(--row-min-height)",
        padding: "12px 16px", borderRadius: "var(--radius-xl)",
        background: divider ? "transparent" : "var(--surface-card)",
        border: divider ? "none" : "var(--border-width) solid var(--border-hairline)",
        borderBottom: divider ? "var(--border-width) solid var(--border-hairline)" : undefined,
        cursor: onClick ? "pointer" : "default", ...style,
      }}
      {...rest}
    >
      {icon && (
        <div style={{ width: 52, height: 52, flex: "0 0 auto", display: "flex", alignItems: "center", justifyContent: "center", borderRadius: iconShape === "circle" ? "var(--radius-pill)" : "var(--radius-md)", ...TILE_TONES[tone] }}>
          <Icon name={icon} size={24} />
        </div>
      )}
      <div style={{ flex: 1, minWidth: 0 }}>
        <div style={{ fontSize: "var(--type-row-title-size)", fontWeight: "var(--type-row-title-weight)", lineHeight: "var(--type-row-title-lh)", color: "var(--text-primary)", overflow: "hidden", textOverflow: "ellipsis", whiteSpace: "nowrap" }}>{title}</div>
        {subtitle && <div style={{ fontSize: "var(--type-body-size)", color: "var(--text-secondary)", marginTop: "3px", overflow: "hidden", textOverflow: "ellipsis", whiteSpace: "nowrap" }}>{subtitle}</div>}
      </div>
      {trailing && <span style={{ fontSize: "16px", fontWeight: "var(--fw-bold)", color: "var(--text-secondary)" }}>{trailing}</span>}
      {chevron && <Icon name="chevron-right" size={20} color="var(--text-tertiary)" />}
    </div>
  );
}
