/** Square or circular icon-only control, minimum 44px tap target. */
export interface IconButtonProps {
  /** Lucide icon name. */
  icon: string;
  /** Accessible label — required, the button has no text. */
  label: string;
  tone?: "neutral" | "accent" | "quiet" | "overlay";
  shape?: "circle" | "square";
  size?: number;
  iconSize?: number;
  onClick?: () => void;
  style?: React.CSSProperties;
}
export declare function IconButton(props: IconButtonProps): JSX.Element;
